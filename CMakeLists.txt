cmake_minimum_required(VERSION 3.20)
project(early_fire_suppression VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(fire_core firmware/src/detection.c firmware/src/suppression.c firmware/src/fire_system.c)
target_include_directories(fire_core PUBLIC firmware/include)
function(fire_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()
fire_warnings(fire_core)
add_executable(fire_demo sim/main.c)
target_link_libraries(fire_demo PRIVATE fire_core)
fire_warnings(fire_demo)
enable_testing()
add_executable(fire_tests tests/test_fire_system.c)
target_link_libraries(fire_tests PRIVATE fire_core)
fire_warnings(fire_tests)
foreach(case_id RANGE 1 20)
    if(case_id LESS 10)
        set(case_name TC-00${case_id})
    else()
        set(case_name TC-0${case_id})
    endif()
    add_test(NAME ${case_name} COMMAND fire_tests ${case_id})
endforeach()
